   148→function leftArm(pose: Pose, g: Geometry): ArmPose {

   155→    case 'cheer':
   156→      return { elbow: [sx - 4, sy - 50], hand: [sx - 18, g.head.cy - 52] };
   157→    case 'point':
   158→      return { elbow: [sx + 8, sy + 40], hand: [sx + 44, sy + 62] };
   159→    case 'hips':
   160→      return { elbow: [sx - 26, sy + 70], hand: [sx + 36, g.hipY - 14] };
   161→    case 'think':
   162→      return { elbow: [sx - 18, sy + 60], hand: [sx + 62, g.chinY - 8] };

   205→function mirror(p: ArmPose, g: Geometry): ArmPose {
   206→  const mx = (x: number) => 512 - x;
   207→  void g;
   208→  return { elbow: [mx(p.elbow[0]), p.elbow[1]], hand: [mx(p.hand[0]), p.hand[1]] };

   213→export function shade(hex: string, amt: number): string {
   214→  const n = hex.replace('#', '');
   215→  const full = n.length === 3 ? n.split('').map((c) => c + c).join('') : n;
   216→  const r = Math.max(0, Math.min(255, parseInt(full.slice(0, 2), 16) + amt));
   217→  const g = Math.max(0, Math.min(255, parseInt(full.slice(2, 4), 16) + amt));
   218→  const b = Math.max(0, Math.min(255, parseInt(full.slice(4, 6), 16) + amt));
   219→  return `rgb(${r},${g},${b})`;
   220→}
   221→
   222→/** Mix a color toward white (amt 0..1) — softer highlights than plain shade() */
   223→function tint(hex: string, amt: number): string {
   224→  const n = hex.replace('#', '');
   225→  const full = n.length === 3 ? n.split('').map((c) => c + c).join('') : n;
   226→  const r = parseInt(full.slice(0, 2), 16);
   227→  const g = parseInt(full.slice(2, 4), 16);
   228→  const b = parseInt(full.slice(4, 6), 16);
   229→  const m = (c: number) => Math.round(c + (255 - c) * amt);
   230→  return `rgb(${m(r)},${m(g)},${m(b)})`;
   231→}

   263→function headPath(h: FaceAnchor): string {
   264→  const { cx, cy, rx, ry } = h;
   265→  const top = cy - ry;
   266→  return [
   267→    `M ${cx} ${top}`,
   268→    `C ${cx + rx * 0.86} ${top + 4} ${cx + rx} ${cy - ry * 0.36} ${cx + rx} ${cy + ry * 0.08}`,
   269→    `C ${cx + rx} ${cy + ry * 0.55} ${cx + rx * 0.6} ${cy + ry} ${cx} ${cy + ry}`,
   270→    `C ${cx - rx * 0.6} ${cy + ry} ${cx - rx} ${cy + ry * 0.55} ${cx - rx} ${cy + ry * 0.08}`,
   271→    `C ${cx - rx} ${cy - ry * 0.36} ${cx - rx * 0.86} ${top + 4} ${cx} ${top} Z`,
   272→  ].join(' ');
   273→}

   275→/** Cylindrical limb: shadow stroke → base stroke → highlight stroke, plus a hand */
   276→function limb(
   277→  sx: number, sy: number, ex: number, ey: number, hx: number, hy: number,
   278→  w: number, p: Palette, part: 'cloth' | 'skin', sleeve: boolean
   279→): string {
   280→  const col = part === 'cloth' ? p.cloth : p.skin;
   281→  const hi = part === 'cloth' ? p.clothHi : p.skinHi;
   282→  const lo = part === 'cloth' ? p.clothLo : p.skinLo;
   283→  const d = `M ${sx} ${sy} Q ${ex} ${ey} ${hx} ${hy}`;
   284→  const handR = w * (sleeve ? 0.5 : 0.58);
   285→  const handFill = sleeve ? p.skin : `url(#gSkin)`;
   286→  return `
   287→    <path d="${d}" stroke="${lo}" stroke-width="${w}" stroke-linecap="round" fill="none" transform="translate(1.6,2)"/>
   288→    <path d="${d}" stroke="${col}" stroke-width="${w}" stroke-linecap="round" fill="none"/>
   289→    <path d="${d}" stroke="${hi}" stroke-width="${Math.max(3, w * 0.34)}" stroke-linecap="round" fill="none" transform="translate(-1.4,-2.2)" opacity="0.75"/>
   290→    <circle cx="${hx}" cy="${hy}" r="${handR}" fill="${handFill}" stroke="${OUTLINE}" stroke-width="1.4"/>
   291→    <circle cx="${hx - handR * 0.28}" cy="${hy - handR * 0.3}" r="${handR * 0.3}" fill="rgba(255,255,255,0.4)"/>`;
   292→}

   374→    <path d="M ${lx} ${sy} Q 256 ${sy - 22} ${rx2} ${sy} L ${256 + tw} ${hipY - 6} Q 256 ${hipY + 10} ${256 - tw} ${hipY - 6} Z" fill="url(#gShadeR)" opacity="0.5"/>

   384→    case 'blazer':

   396→        <path d="M ${256 - tw - 8} ${hipY + 96} Q 256 ${hipY + 120} ${256 + tw + 8} ${hipY + 96}" stroke="rgba(255,255,255,0.35)" stroke-width="6" fill="none"/>
   397→        ${volume}`;
   398→    case 'tank':
   399→      return `

   418→/** legs with walk-cycle stride support */
   419→function legs(look: CharacterLook, g: Geometry, pose: Pose, p: Palette): string {
   420→  const pc = p.pants;
   421→  const lw = Math.round(g.hipHalfW * 0.52);
   422→  const shoe = (x: number, y: number, rot = 0) => `
   423→    <g transform="rotate(${rot} ${x} ${y})">
   424→      <ellipse cx="${x}" cy="${y}" rx="30" ry="14" fill="url(#gShoe)" stroke="${OUTLINE}" stroke-width="1.5"/>
   425→      <ellipse cx="${x}" cy="${y + 7}" rx="28" ry="7" fill="${p.shoeHi}" opacity="0.55"/>
   426→      <ellipse cx="${x - 8}" cy="${y - 6}" rx="9" ry="4.5" fill="rgba(255,255,255,0.3)"/>
   427→    </g>`;
   428→  const pantLeg = (x1: number, y1: number, x2: number, y2: number) => `
   429→    <path d="M ${x1} ${y1} L ${x2} ${y2}" stroke="${p.pantsLo}" stroke-width="${lw}" stroke-linecap="round" fill="none" transform="translate(1.6,1.4)"/>
   430→    <path d="M ${x1} ${y1} L ${x2} ${y2}" stroke="${pc}" stroke-width="${lw}" stroke-linecap="round" fill="none"/>
   431→    <path d="M ${x1} ${y1} L ${x2} ${y2}" stroke="${p.pantsHi}" stroke-width="${Math.max(4, lw * 0.3)}" stroke-linecap="round" fill="none" transform="translate(-1.6,-2)" opacity="0.6"/>`;
   432→
   433→  if (pose === 'sit' || pose === 'pet') {
   434→    const kneeY = g.hipY + 26;
   435→    return `
   436→      ${pantLeg(256 - 34, g.hipY, 256 + 60, kneeY)}
   437→      ${pantLeg(256 - 10, g.hipY + 4, 256 + 82, kneeY + 10)}
   438→      ${pantLeg(256 + 60, kneeY, 256 + 60, g.footY - 26)}
   439→      ${pantLeg(256 + 82, kneeY + 10, 256 + 82, g.footY - 18)}
   440→      ${shoe(256 + 74, g.footY - 10)}
   441→      ${shoe(256 + 96, g.footY - 6)}
   442→      <rect x="${256 - g.hipHalfW - 6}" y="${g.hipY - 18}" width="${g.hipHalfW * 2 + 12}" height="34" rx="14" fill="${pc}"/>`;
   443→  }
   444→  if (pose === 'walkA') {
   445→    // left leg forward, right leg back
   446→    const lx = 256 - g.hipHalfW + lw / 2;
   447→    const rx2 = 256 + g.hipHalfW - lw / 2;
   448→    return `
   449→      ${pantLeg(rx2, g.legTopY - 12, rx2 - 30, g.footY - 22)}
   450→      ${pantLeg(lx, g.legTopY - 12, lx + 26, g.footY - 12)}
   451→      ${shoe(rx2 - 34, g.footY - 14, -14)}
   452→      ${shoe(lx + 32, g.footY - 4, 8)}`;
   453→  }
   454→  if (pose === 'walkB') {
   455→    const lx = 256 - g.hipHalfW + lw / 2;
   456→    const rx2 = 256 + g.hipHalfW - lw / 2;
   457→    return `
   458→      ${pantLeg(lx, g.legTopY - 12, lx - 26, g.footY - 18)}
   459→      ${pantLeg(rx2, g.legTopY - 12, rx2 + 26, g.footY - 12)}
   460→      ${shoe(lx - 30, g.footY - 10, -10)}
   461→      ${shoe(rx2 + 32, g.footY - 4, 10)}`;
   462→  }
   463→  const lx = 256 - g.hipHalfW + lw / 2;
   464→  const rx2 = 256 + g.hipHalfW - lw / 2;
   465→  return `
   466→    ${pantLeg(lx, g.legTopY - 12, lx, g.footY - 14)}
   467→    ${pantLeg(rx2, g.legTopY - 12, rx2, g.footY - 14)}
   468→    ${shoe(lx, g.footY - 4)}
   469→    ${shoe(rx2 + 6, g.footY - 4)}`;
   470→}

   474→  switch (pose) {

   496→  }

   500→/** Builds the full character sprite SVG (face features are drawn by the engine) */
   501→export function buildCharacterSvg(look: CharacterLook, pose: Pose = 'idle'): string {
   502→  const g = geometry(look);
   503→  const p = palette(look);
   504→  const armL = leftArm(pose, g);
   505→  const armR = mirror(armL, g);
   506→  const h = g.head;
   507→
   508→  const sleeveW = Math.round(g.armW * 1.12);
   509→  const sleeve = (ap: ArmPose) =>
   510→    limb(g.shoulderLX + 4, g.shoulderY + 6, ap.elbow[0], ap.elbow[1], ap.hand[0], ap.hand[1], sleeveW, p, 'cloth', true);
   511→  const armRSleeve = (ap: ArmPose) => {
   512→    const m = mirror(ap, g);
   513→    return limb(512 - (g.shoulderLX + 4), g.shoulderY + 6, m.elbow[0], m.elbow[1], m.hand[0], m.hand[1], sleeveW, p, 'cloth', true);
   514→  };
   515→
   516→  const neck = `
   517→    <rect x="${256 - g.neckW / 2}" y="${g.neckTopY}" width="${g.neckW}" height="${g.shoulderY - g.neckTopY + 10}" rx="${g.neckW / 3}" fill="url(#gNeck)"/>
   518→    <ellipse cx="256" cy="${g.neckTopY + 5}" rx="${g.neckW * 0.62}" ry="7" fill="${p.skinLo}" opacity="0.55"/>`;
   519→
   520→  const head = `
   521→    <g>
   522→      <!-- ears -->
   523→      <ellipse cx="${h.cx - h.rx - 3}" cy="${h.cy + h.ry * 0.12}" rx="${h.rx * 0.16}" ry="${h.rx * 0.24}" fill="url(#gSkin)" stroke="${OUTLINE}" stroke-width="1.3"/>
   524→      <ellipse cx="${h.cx + h.rx + 3}" cy="${h.cy + h.ry * 0.12}" rx="${h.rx * 0.16}" ry="${h.rx * 0.24}" fill="url(#gSkin)" stroke="${OUTLINE}" stroke-width="1.3"/>
   525→      <!-- face volume -->
   526→      <path d="${headPath(h)}" fill="url(#gSkin)" stroke="${OUTLINE}" stroke-width="1.8"/>
   527→      <!-- jaw / cheek shading -->
   528→      <path d="M ${h.cx - h.rx * 0.72} ${h.cy + h.ry * 0.42}
   529→               C ${h.cx - h.rx * 0.4} ${h.cy + h.ry * 0.88} ${h.cx + h.rx * 0.4} ${h.cy + h.ry * 0.88} ${h.cx + h.rx * 0.72} ${h.cy + h.ry * 0.42}"
   530→            stroke="${p.skinLo}" stroke-width="${h.rx * 0.1}" stroke-linecap="round" fill="none" opacity="0.35"/>
   531→      <!-- forehead highlight -->
   532→      <ellipse cx="${h.cx - h.rx * 0.3}" cy="${h.cy - h.ry * 0.5}" rx="${h.rx * 0.42}" ry="${h.ry * 0.2}" fill="rgba(255,255,255,0.28)"/>
   533→    </g>`;
   534→
   535→  const figure = `
   536→    ${legs(look, g, pose, p)}
   537→    ${armRSleeve(armR)}
   538→    ${sleeve(armL)}
   539→    ${neck}
   540→    ${outfitTorso(look, g, p)}
   541→    ${hairShapes(look, g, p)}
   542→    ${head}
   543→    ${poseProps(pose, g, p)}
   544→  `;
   545→
   546→  return `<svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 ${SPRITE_W} ${SPRITE_H}" width="${SPRITE_W}" height="${SPRITE_H}">
   547→  <defs>
   548→    <radialGradient id="gSkin" cx="0.38" cy="0.28" r="0.95">
   549→      <stop offset="0" stop-color="${p.skinHi}"/>
   550→      <stop offset="0.52" stop-color="${p.skin}"/>
   551→      <stop offset="1" stop-color="${p.skinLo}"/>
   552→    </radialGradient>
   553→    <linearGradient id="gNeck" x1="0" y1="0" x2="0" y2="1">
   554→      <stop offset="0" stop-color="${p.skinLo}"/>
   555→      <stop offset="0.6" stop-color="${p.skin}"/>
   556→      <stop offset="1" stop-color="${p.skinHi}"/>
   557→    </linearGradient>
   558→    <linearGradient id="gCloth" x1="0" y1="0" x2="0" y2="1">
   559→      <stop offset="0" stop-color="${p.clothHi}"/>
   560→      <stop offset="0.45" stop-color="${p.cloth}"/>
   561→      <stop offset="1" stop-color="${p.clothLo}"/>
   562→    </linearGradient>
   563→    <linearGradient id="gShirt" x1="0" y1="0" x2="0" y2="1">
   564→      <stop offset="0" stop-color="#F2EEE6"/>
   565→      <stop offset="1" stop-color="#C9C2B4"/>
   566→    </linearGradient>
   567→    <linearGradient id="gHair" x1="0" y1="0" x2="0" y2="1">
   568→      <stop offset="0" stop-color="${p.hairHi}"/>
   569→      <stop offset="0.5" stop-color="${p.hair}"/>
   570→      <stop offset="1" stop-color="${p.hairLo}"/>
   571→    </linearGradient>
   572→    <linearGradient id="gShoe" x1="0" y1="0" x2="0" y2="1">
   573→      <stop offset="0" stop-color="${p.shoeHi}"/>
   574→      <stop offset="1" stop-color="${p.shoe}"/>
   575→    </linearGradient>
   576→    <linearGradient id="gShadeR" x1="0" y1="0" x2="1" y2="0">
   577→      <stop offset="0" stop-color="rgba(40,26,20,0)"/>
   578→      <stop offset="1" stop-color="rgba(40,26,20,0.35)"/>
   579→    </linearGradient>
   580→  </defs>
   581→  <g>${figure}</g>
   582→</svg>`;
   583→}

   585→/** stable cache key for a character look + pose */
   586→export function lookKey(look: CharacterLook, pose: Pose = 'idle'): string {
   587→  const s = `v3|${look.skin}|${look.hair}|${look.hairColor}|${look.outfit}|${look.outfitColor}|${look.pantsColor}|${look.glasses ? 1 : 0}|${look.irisColor ?? ''}|${look.age ?? 'adult'}|${pose}`;
   588→  let h = 5381;
   589→  for (let i = 0; i < s.length; i++) h = ((h << 5) + h + s.charCodeAt(i)) >>> 0;
   590→  return `char-${h.toString(36)}`;
   591→}
